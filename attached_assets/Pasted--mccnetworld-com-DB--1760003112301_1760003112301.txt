파일: server/db.ts

1) `checkPostgreSQLHealth`(이름이 다를 수 있음) 함수에서 DB 상태를 조회하는 쿼리를 다음 가이드로 수정:
- 기존(문제): `current_setting('ssl') as ssl_is_on` → 일부 환경에서 예외 발생 시 전체 실패
- 수정(안전):
  - 1안: NULL 안전 파라미터 사용 후 COALESCE
    ```sql
    COALESCE(current_setting('ssl', true), 'off') as ssl_is_on
    ```
  - 2안(보강): SSL 조회 블록만 별도 try/catch로 감싸고, 실패 시 `'unknown'`으로 대체

2) 타입스크립트 예시(참고용) — 쿼리 템플릿은 프로젝트 DB 클라이언트(pg/prisma/drizzle)에 맞게 적용:
```ts
export async function checkPostgreSQLHealth(db: any) {
  try {
    // 핵심: SSL 설정 조회가 실패해도 전체 헬스체크는 통과해야 함
    const sql = `
      SELECT
        1 as health_check,
        inet_server_addr()::text as host,
        inet_server_port() as port,
        current_database() as db,
        current_user as user,
        COALESCE(current_setting('ssl', true), 'off') as ssl_is_on
    `;
    const row = (await db.query(sql)).rows?.[0];

    return {
      ok: true,
      host: row?.host ?? null,
      port: row?.port ?? null,
      db: row?.db ?? null,
      user: row?.user ?? null,
      ssl: row?.ssl_is_on ?? 'off',
    };
  } catch (e) {
    // 만약 위 쿼리가 환경에 따라 실패하면, 최소 헬스만 재확인(폴백)
    try {
      const minimal = (await db.query(`SELECT 1 as health_check`)).rows?.[0];
      if (minimal?.health_check === 1) {
        return { ok: true, host: null, port: null, db: null, user: null, ssl: 'unknown' };
      }
    } catch {}
    return { ok: false, error: String(e) };
  }
}
