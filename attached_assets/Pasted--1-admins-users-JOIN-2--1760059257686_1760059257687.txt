import { admins, users, documents, contactCodes } from "@/server/db/schema"; // 실제 경로에 맞게 유지
----------------------------------------------------------------

----------------------------------------------------------------
// 기존(예: users만 JOIN)

// 수정: admins + users 모두 LEFT JOIN, 관리자 이름 우선
if (filters?.includeActivatedBy) {
  query = db.select({
    ...docColumns,
    dealerName: contactCodes.dealerName,
    activatedByName: sql<string>`COALESCE(${admins.name}, ${users.name})`,
  })
  .from(documents)
  .leftJoin(contactCodes, eq(documents.contactCodeId, contactCodes.id))
  .leftJoin(admins, eq(documents.activatedBy, admins.id))
  .leftJoin(users, eq(documents.activatedBy, users.id));
}
----------------------------------------------------------------

2) 프론트엔드: client/src/pages/SubmitApplication.tsx
- 제출 버튼과 onSubmit 핸들러에 “중복 제출 가드” 적용
- 아래 패턴을 참고해 현재 파일의 동일 위치에 반영(이미 uploadMutation이 있다면 그대로 사용)

----------------------------------------------------------------
/** 가드 추가: 진행 중에는 중복 제출 차단 */
const handleSubmit = (e: React.FormEvent<HTMLFormElement>) => {
  e.preventDefault();
  if (uploadMutation.isPending /* 또는 isLoading */) return; // 🔒 진행 중 가드

  const formData = new FormData(e.currentTarget);
  // ... 기존 유효성 검사/데이터 준비
  uploadMutation.mutate(formData);
};

/** 버튼 비활성화: 진행 중일 때 클릭 불가 */
<Button type="submit" onClick={() => {}} disabled={uploadMutation.isPending}>
  접수 신청
</Button>
----------------------------------------------------------------

- 만약 onSubmit을 form에 바인딩하고 있다면 form에도 동일 가드를 적용
----------------------------------------------------------------
<form onSubmit={handleSubmit}>
  {/* ...fields... */}
  <Button type="submit" disabled={uploadMutation.isPending} aria-busy={uploadMutation.isPending}>
    접수 신청
  </Button>
</form>
----------------------------------------------------------------

- (선택 권장) 중복 전송을 더 강하게 막기 위한 추가 보강
  · 버튼에 keydown/keyup으로 엔터 연타 방지(필요 시)
  · 요청에 idempotency 키를 헤더로 첨부(백엔드에서 같은 키의 중복 요청 무시)
  예) 헤더 추가:
  ----------------------------------------------------------------
  uploadMutation.mutate(formData, {
    onMutate() {
      // ex) window.crypto.randomUUID()
      formData.append("X-Idempotency-Key", crypto.randomUUID());
    }
  });
  ----------------------------------------------------------------
  (백엔드에서 해당 키를 메모리/DB에 기록해 일정 시간 내 동일 키 중복 생성 차단 — 별도 변경으로 분리)
