[1] DB 스키마 — shared/schema.ts (Drizzle/Postgres)
----------------------------------------------------------------
import { pgTable, serial, integer, text, boolean, timestamp, varchar, index, unique, sql } from "drizzle-orm/pg-core";
import { users } from "./users"; // 실제 경로에 맞게

export const announcements = pgTable("announcements", {
  id: serial("id").primaryKey(),
  title: varchar("title", { length: 200 }).notNull(),
  content: text("content").notNull(),
  authorId: integer("author_id").notNull().references(() => users.id, { onDelete: "cascade" }),

  targetType: varchar("target_type", { length: 20 }).notNull(), // 'all' | 'carrier' | 'users'
  targetCarriers: text("target_carriers").array().default(sql`ARRAY[]::text[]`), // 예: ['SKT','KT']
  targetUserIds: integer("target_user_ids").array().default(sql`ARRAY[]::int[]`),

  isActive: boolean("is_active").notNull().default(true),
  scheduledAt: timestamp("scheduled_at", { withTimezone: true, mode: "string" }).default(sql`NOW()`),
  expiresAt: timestamp("expires_at", { withTimezone: true, mode: "string" }).nullable(),

  createdAt: timestamp("created_at", { withTimezone: true, mode: "string" }).defaultNow(),
  updatedAt: timestamp("updated_at", { withTimezone: true, mode: "string" }).defaultNow(),
}, (t) => ({
  idxActiveSched: index("ann_active_sched_idx").on(t.isActive, t.scheduledAt),
}));

export const announcementReads = pgTable("announcement_reads", {
  id: serial("id").primaryKey(),
  announcementId: integer("announcement_id").notNull().references(() => announcements.id, { onDelete: "cascade" }),
  userId: integer("user_id").notNull().references(() => users.id, { onDelete: "cascade" }),
  readAt: timestamp("read_at", { withTimezone: true, mode: "string" }).defaultNow(),
}, (t) => ({
  uniqRead: unique().on(t.announcementId, t.userId),
  idxUser: index("ann_reads_user_idx").on(t.userId),
  idxAnn: index("ann_reads_ann_idx").on(t.announcementId),
}));

----------------------------------------------------------------
import { and, or, sql, desc, eq, notInArray } from "drizzle-orm";
import { announcements, announcementReads } from "@/shared/schema";

type CreateAnn = {
  title: string; content: string; authorId: number;
  targetType: 'all'|'carrier'|'users';
  targetCarriers?: string[]; // 표준 키
  targetUserIds?: number[];
  isActive?: boolean;
  scheduledAt?: string|null;
  expiresAt?: string|null;
};

export async function createAnnouncement(data: CreateAnn) {
  // 정상화
  const carriers = (data.targetCarriers ?? []).map(normalizeCarrier);
  const userIds = Array.from(new Set(data.targetUserIds ?? []));

  return await db.insert(announcements).values({
    title: data.title.trim(),
    content: data.content.trim(),
    authorId: data.authorId,
    targetType: data.targetType,
    targetCarriers: carriers,
    targetUserIds: userIds,
    isActive: data.isActive ?? true,
    scheduledAt: data.scheduledAt ?? sql`NOW()`,
    expiresAt: data.expiresAt ?? null,
  }).returning();
}

export async function updateAnnouncement(id: number, patch: Partial<CreateAnn> & {isActive?: boolean}) {
  // 필요한 키만 업데이트, 배열은 중복 제거/정규화
  const upd: any = {};
  if (patch.title !== undefined) upd.title = patch.title.trim();
  if (patch.content !== undefined) upd.content = patch.content.trim();
  if (patch.targetType) upd.targetType = patch.targetType;
  if (patch.targetCarriers) upd.targetCarriers = Array.from(new Set(patch.targetCarriers.map(normalizeCarrier)));
  if (patch.targetUserIds) upd.targetUserIds = Array.from(new Set(patch.targetUserIds));
  if (patch.isActive !== undefined) upd.isActive = patch.isActive;
  if (patch.scheduledAt !== undefined) upd.scheduledAt = patch.scheduledAt;
  if (patch.expiresAt !== undefined) upd.expiresAt = patch.expiresAt;
  upd.updatedAt = sql`NOW()`;

  return await db.update(announcements).set(upd).where(eq(announcements.id, id)).returning();
}

export async function deleteAnnouncement(id: number) {
  // 소프트 삭제(비활성) 권장
  return await db.update(announcements).set({ isActive: false, updatedAt: sql`NOW()` })
    .where(eq(announcements.id, id)).returning();
}

export async function getAnnouncements({ activeOnly=false, limit=50, offset=0 } = {}) {
  const where = activeOnly
    ? and(eq(announcements.isActive, true))
    : undefined;
  return await db.select().from(announcements)
    .where(where).orderBy(desc(announcements.createdAt)).limit(limit).offset(offset);
}

export async function markAnnouncementAsRead(announcementId: number, userId: number) {
  await db.execute(sql`
    INSERT INTO "announcement_reads" ("announcement_id","user_id")
    VALUES (${announcementId}, ${userId})
    ON CONFLICT ("announcement_id","user_id") DO NOTHING
  `);
}

export async function getUnreadAnnouncements(userId: number) {
  // 사용자 캐리어/역할 조회 (실제 구현의 사용자 테이블/필드에 맞게)
  const u = await getUserById(userId); // { carrier: 'SKT' | ... }
  const userCarrier = normalizeCarrier(u?.carrier ?? '');

  // 이미 읽은 공지 제외
  const readIds = await db.select({ id: announcementReads.announcementId })
    .from(announcementReads)
    .where(eq(announcementReads.userId, userId));

  // 필터: 활성 + 스케줄 범위 + (ALL or CARRIER contain or USERS contain)
  return await db.select().from(announcements).where(and(
    eq(announcements.isActive, true),
    or(sql`(${announcements.scheduledAt} IS NULL)`, sql`${announcements.scheduledAt} <= NOW()`),
    or(sql`(${announcements.expiresAt} IS NULL)`, sql`${announcements.expiresAt} >= NOW()`),
    notInArray(announcements.id, readIds.map(r => r.id)),
    or(
      eq(announcements.targetType, 'all'),
      and(
        eq(announcements.targetType, 'carrier'),
        // target_carriers @> ARRAY[userCarrier]
        sql`(${announcements.targetCarriers}) @> ARRAY[${userCarrier}]::text[]`
      ),
      and(
        eq(announcements.targetType, 'users'),
        // target_user_ids @> ARRAY[userId]
        sql`(${announcements.targetUserIds}) @> ARRAY[${userId}]::int[]`
      )
    )
  ))
  .orderBy(desc(announcements.scheduledAt), desc(announcements.createdAt));
}

export async function getAnnouncementStats(announcementId: number) {
  const base = await db.select().from(announcements).where(eq(announcements.id, announcementId)).limit(1);
  if (!base.length) return null;
  const a = base[0];

  // 대상 총원 계산
  let targetCount = 0;
  if (a.targetType === 'all') {
    targetCount = await getTotalUserCount();
  } else if (a.targetType === 'carrier') {
    for (const c of (a.targetCarriers ?? [])) {
      targetCount += await getUserCountByCarrier(c);
    }
  } else if (a.targetType === 'users') {
    targetCount = (a.targetUserIds ?? []).length;
  }

  const readCountRow = await db.execute(sql`
    SELECT COUNT(*)::int AS cnt FROM "announcement_reads" WHERE "announcement_id" = ${announcementId}
  `);
  const read = Number(readCountRow.rows?.[0]?.cnt ?? 0);

  return { target: targetCount, read, unread: Math.max(targetCount - read, 0) };
}

// helpers
function normalizeCarrier(x: string) {
  const s = (x || '').trim().toUpperCase().replace(/\s+/g,'');
  if (s === 'LGU+' || s === 'LGUPLUS' || s === 'LGU') return 'LGU+';
  if (s === 'SK' || s === 'SKT') return 'SKT';
  if (s === 'KT') return 'KT';
  return s || 'MVNO';
}

[6] 검증 시나리오
----------------------------------------------------------------
1) 전체 발송: targetType=all → 모든 사용자 로그인 시 1회 팝업
2) 통신망 발송: targetType=carrier, ['SKT','KT'] → SKT/KТ 사용자에만 노출
3) 개별 발송: targetType=users, [5,12,23] → 해당 3명에게만 노출
4) 읽음 처리: 같은 사용자가 재로그인해도 동일 공지 미노출 (reads UNIQUE 보장)
5) 통계: stats API에서 대상/읽음/미읽음 수 일치
