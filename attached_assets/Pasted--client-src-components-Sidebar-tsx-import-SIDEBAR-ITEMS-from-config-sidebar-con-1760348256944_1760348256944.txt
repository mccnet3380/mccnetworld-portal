사이드바 컴포넌트 수정: client/src/components/Sidebar.tsx

import { SIDEBAR_ITEMS } from '@/config/sidebar.config';
const APP_ENV = import.meta.env.VITE_APP_ENV || (process.env.APP_ENV ?? 'development');
const FLAGS = (window as any).__FLAGS__ || {}; // 필요 시 전역 플래그 주입

const items = SIDEBAR_ITEMS.filter(it => {
  const envOk  = !it.env   || it.env.includes(APP_ENV as any);
  const roleOk = !it.roles || it.roles.includes(currentUser.role);
  const flagOk = !it.flag  || FLAGS[it.flag] === true;
  return envOk && roleOk && flagOk;
});

// i18n으로 렌더
return (
  <nav>
    {items.map(item => (
      <Link key={item.key} to={item.path}>{t(item.key)}</Link>
    ))}
  </nav>
);

[5] 배포 변수

개발: APP_ENV=development, 운영 Publish: APP_ENV=production

VITE_APP_ENV를 빌드 시 주입(또는 서버에서 window.FLAGS 주입)

검증 시나리오(수용 기준)

운영 빌드에서 개발 전용 메뉴(예: 관리자 패널)가 보이지 않는다.

개발 빌드에서는 개발 전용 메뉴가 보이고 정상 진입된다.

ts-prune/depcheck/unused-imports 검사 통과 후에만 빌드된다(미사용 코드가 있으면 빌드 실패).

/_archive의 내용이 운영 번들에 포함되지 않는다(임포트 금지).

“관리자/관리자 패널” 등 명칭은 i18n 키로만 관리되며, 문자열 하드코딩이 사라진다.
