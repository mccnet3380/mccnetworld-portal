3) 통계 쿼리 수정 — child 제외
- getMonthlyActivationSummary, getTodayStats 내부에서 집계 대상을
  “child 문서 제외(= 부모/대표 row만)”로 제한.
- 텍스트 notes 기준이므로 LIKE 사용.

예시 A) 월간 개통수량 (개념)
----------------------------------------------------------------
export async function getMonthlyActivationSummary(db, y: number, m: number) {
  const monthStart = new Date(Date.UTC(y, m - 1, 1, 0, 0, 0));
  const monthEnd   = new Date(Date.UTC(y, m, 0, 23, 59, 59));

  // activationStatus 가 '개통'(또는 프로젝트 상 정의된 완료 상태) 조건은 기존 그대로
  // ✅ child 제외: notes NOT LIKE CHILD_LIKE
  const rows = await db
    .select({ id: documents.id })
    .from(documents)
    .where(and(
      gte(documents.activatedAt, monthStart),
      lte(documents.activatedAt, monthEnd),
      sql`${documents.activationStatus} = '개통'`,
      sql`${documents.notes} NOT LIKE ${CHILD_LIKE}`
    ));

  // 부모 row만 카운트 → 중복(3장=3건) 제거
  const count = rows.length;

  logBatch("MonthlySummary", { y, m, count });
  return { count };
}

예시 B) 당일 현황 (개념)

export async function getTodayStats(db, tzNow?: Date) {
  const now = tzNow ?? new Date();
  const start = new Date(Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate(), 0,0,0));
  const end   = new Date(Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate(), 23,59,59));

  const rows = await db
    .select({ id: documents.id })
    .from(documents)
    .where(and(
      gte(documents.activatedAt, start),
      lte(documents.activatedAt, end),
      sql`${documents.activationStatus} = '개통'`,
      sql`${documents.notes} NOT LIKE ${CHILD_LIKE}`
    ));

  const count = rows.length;
  logBatch("TodayStats", { date: start.toISOString().slice(0,10), count });
  return { count };
}
----------------------------------------------------------------
