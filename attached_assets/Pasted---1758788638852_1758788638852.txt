// 테이블 존재 여부 확인: other_business_carriers 가드용
private async tableExists(tableName: string): Promise<boolean> {
  const db = await this.ensureDatabase();
  try {
    const result = await db.execute(sql`
      SELECT 1
      FROM information_schema.tables
      WHERE table_schema = 'public'
        AND table_name = ${tableName}
      LIMIT 1
    `);
    // drizzle-pg: rows 혹은 length 유무로 판단
    // @ts-ignore
    return !!(result?.rows?.length || result?.length);
  } catch {
    return false;
  }
}

B) getTodayStats() 메서드 전체 교체
async getTodayStats() {
  await this.ensureDatabase();

  // 오늘 00:00 ~ 현재
  const now = new Date();
  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate());

  // 당일 개통(activation_status='개통')
  const totalToday = await this.db.execute(sql`
    SELECT COUNT(*)::int AS cnt
    FROM documents
    WHERE activation_status = '개통'
      AND activated_at >= ${startOfToday} AND activated_at <= ${now}
  `);

  // 신규/번호이동 분리 집계 (customer_type: 'new' | 'port_in')
  const typeStats = await this.db.execute(sql`
    SELECT customer_type, COUNT(*)::int AS cnt
    FROM documents
    WHERE activation_status = '개통'
      AND activated_at >= ${startOfToday} AND activated_at <= ${now}
      AND customer_type IN ('new', 'port_in')
    GROUP BY customer_type
  `);

  // 캐리어 별 당일 개통 집계
  const carrierStats = await this.db.execute(sql`
    SELECT carrier, COUNT(*)::int AS cnt
    FROM documents
    WHERE activation_status = '개통'
      AND activated_at >= ${startOfToday} AND activated_at <= ${now}
    GROUP BY carrier
    ORDER BY cnt DESC
  `);

  // 기타업무(선택): 테이블 없으면 빈 배열로 반환해 운영 안정성 유지
  let otherBusinessToday = 0;
  if (await this.tableExists('other_business_carriers')) {
    const otherRes = await this.db.execute(sql`
      SELECT COUNT(*)::int AS cnt
      FROM other_business_carriers
      WHERE created_at >= ${startOfToday} AND created_at <= ${now}
    `);
    // @ts-ignore
    otherBusinessToday = otherRes?.rows?.[0]?.cnt ?? otherRes?.[0]?.cnt ?? 0;
  }

  // @ts-ignore
  const total = totalToday?.rows?.[0]?.cnt ?? totalToday?.[0]?.cnt ?? 0;

  // @ts-ignore
  const newCount = (typeStats?.rows ?? typeStats ?? []).find((r:any) => r.customer_type === 'new')?.cnt ?? 0;
  // @ts-ignore
  const portInCount = (typeStats?.rows ?? typeStats ?? []).find((r:any) => r.customer_type === 'port_in')?.cnt ?? 0;

  // @ts-ignore
  const carriers = (carrierStats?.rows ?? carrierStats ?? []).map((r:any) => ({
    carrier: r.carrier,
    count: r.cnt
  }));

  return {
    totalActivationsToday: total,
    newActivationsToday: newCount,
    portInActivationsToday: portInCount,
    otherBusinessToday,
    carrierStats: carriers
  };
}


위 메서드는 documents(activated_at, activation_status, customer_type, carrier) 기준으로 집계합니다. 컬럼명이 프로젝트와 다르면 컬럼명만 맞춰주세요(스키마 변경 없이).
