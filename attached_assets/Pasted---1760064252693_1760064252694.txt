function formatBytes(bytes?: number): string {
  if (!bytes && bytes !== 0) return "";
  const sizes = ["B","KB","MB","GB","TB"];
  const i = bytes === 0 ? 0 : Math.floor(Math.log(bytes) / Math.log(1024));
  const val = bytes / Math.pow(1024, i);
  return `${val % 1 === 0 ? val : val.toFixed(1)} ${sizes[i]}`;
}

type AnyFile = {
  id?: string|number;
  name?: string;
  filename?: string;
  size?: number;
  fileSize?: number;
  url?: string;
  downloadUrl?: string;
  path?: string;
};

function normalizeFiles(doc: any): AnyFile[] {
  // 1) attachments 배열 우선
  if (Array.isArray(doc?.attachments) && doc.attachments.length > 0) {
    return doc.attachments as AnyFile[];
  }
  // 2) 단일 필드 폴백(attachment/file/attachedFile 등)
  const single = doc?.attachment ?? doc?.file ?? doc?.attachedFile;
  if (single) return [single as AnyFile];
  // 3) 최종 폴백 없음
  return [];
}

function fileDisplayName(f: AnyFile): string {
  return (f.name ?? f.filename ?? `file-${f.id ?? ""}`).toString();
}

function fileHref(f: AnyFile, doc: any): string | undefined {
  // 우선순위: url > downloadUrl > path
  return f.url ?? f.downloadUrl ?? f.path;
}

3) 테이블/리스트의 "첨부파일" 셀 렌더링 수정
- 기존에 단일 파일만 가정하고 렌더링하던 코드를 찾아 아래 패턴으로 교체
- 테이블 셀 내부 예시:

----------------------------------------------------------------
// 기존 (예시)
// <td>{doc.attachment?.name}</td>

// 수정
<td>
  {(() => {
    const files = normalizeFiles(doc);
    if (files.length === 0) {
      // 기존 동작 유지: 아무것도 없으면 빈칸 또는 기존 단일 표현
      return doc?.attachment?.name ?? "-";
    }
    return (
      <ul style={{ margin: 0, paddingLeft: "1rem" }}>
        {files.map((f, idx) => {
          const href = fileHref(f, doc);
          const size = formatBytes(f.size ?? f.fileSize);
          const label = fileDisplayName(f);
          return (
            <li key={f.id ?? `${doc.id}-att-${idx}`}>
              {href ? (
                <a href={href} target="_blank" rel="noopener noreferrer" download>
                  {label}
                </a>
              ) : (
                <span>{label}</span>
              )}
              {size ? <span style={{ opacity: 0.7, marginLeft: 6 }}>({size})</span> : null}
            </li>
          );
        })}
      </ul>
    );
  })()}
</td>
