2) 운영 DB 백업 + 백필 실행 (트랜잭션)

반드시 트랜잭션으로 실행하고, 먼저 백업하세요.

-- 🔒 트랜잭션 시작
BEGIN;

-- 2-0) 즉시 백업 (가벼운 스냅샷)
CREATE TABLE IF NOT EXISTS backup_users_YYYYMMDD AS
  SELECT * FROM users;

CREATE TABLE IF NOT EXISTS backup_documents_YYYYMMDD AS
  SELECT * FROM documents;

-- 2-1) users.dealer_id 매핑 (대표명 기준)
UPDATE users u
SET dealer_id = dr.id
FROM dealer_registrations dr
WHERE u.dealer_id IS NULL
  AND (u.user_type = 'user' OR u.user_type = 'dealer')
  AND LOWER(TRIM(u.name)) = LOWER(TRIM(dr.representative_name));

-- 2-2) documents.dealer_id 백필 (users 기준)
UPDATE documents d
SET dealer_id = u.dealer_id
FROM users u
WHERE d.user_id = u.id
  AND d.dealer_id IS NULL
  AND u.dealer_id IS NOT NULL;

-- 2-3) 결과 점검 (눈으로 확인)
SELECT 
  (SELECT COUNT(*) FROM users WHERE user_type IN ('user','dealer') AND dealer_id IS NULL) AS users_null_dealer,
  (SELECT COUNT(*) FROM documents WHERE dealer_id IS NULL) AS docs_null_dealer;

-- 문제가 없으면 커밋
COMMIT;
-- 이상 있으면 ROLLBACK;


대표명 기준 매핑이 애매한 계정은 수동 매핑 리스트로 별도 UPDATE 하시면 됩니다. (예: UPDATE users SET dealer_id=3 WHERE id=17;)

SELECT id, username, name

-- dealer_id 없는 최근 문서(7일)
SELECT id, user_id, dealer_id, dealer_name_snapshot, contact_code_snapshot, created_at
FROM documents
WHERE dealer_id IS NULL

ORDER BY id DESC;
